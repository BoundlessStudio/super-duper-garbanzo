# TanStack Migration Plan: CustomWare Portal

## Executive Summary

This document outlines a comprehensive migration plan to convert the CustomWare Portal from its current **react-router-dom v7 + Zustand** architecture to **TanStack Router + TanStack DB**. The migration will improve type safety, developer experience, and provide a foundation for future server-side data synchronization.

---

## Current Architecture Analysis

### Technology Stack
| Layer | Current | Target |
|-------|---------|--------|
| **Framework** | React 19 | React 19 (no change) |
| **Build Tool** | Vite 7 | Vite 7 (no change) |
| **Routing** | react-router-dom v7 | TanStack Router v1 |
| **State/Data** | Zustand + localStorage | TanStack DB |
| **Data Fetching** | None (React Query installed unused) | TanStack Query (optional) |
| **Styling** | Tailwind CSS v4 | Tailwind CSS v4 (no change) |

### Current Route Structure
```
/                 -> Projects (dashboard)
/new              -> CreateProject
/:id              -> ProjectDetail
```

### Current Data Model
```typescript
interface Project {
  id: string;
  name: string;
  tasks: Task[];
  settings: ProjectSettings;
  createdAt: string;
  updatedAt: string;
}

interface Task {
  id: string;
  title: string;
  description: string;
  status: 'todo' | 'in-progress' | 'done';
  priority: 'low' | 'medium' | 'high';
  completed: boolean;
  createdAt: string;
}
```

```typescript
// Task schema with validation
export const taskSchema = z.object({

  projectId: z.string(),
  title: z.string().min(1),
  description: z.string(),
  status: z.enum(['todo', 'in-progress', 'done']),

  completed: z.boolean(),
  createdAt: z.string(),
})

// Project schema with validation
export const projectSchema = z.object({
  id: z.string(),
  name: z.string().min(1),
  createdAt: z.string(),
  updatedAt: z.string(),
})

// Project settings schema
export const projectSettingsSchema = z.object({
  projectId: z.string(),
  agentEnabled: z.boolean(),
  agentModel: z.string(),
  agentTemperature: z.number(),
  agentMaxTokens: z.number(),
  githubEnabled: z.boolean(),
  githubRepo: z.string(),
  githubBranch: z.string(),
  sandboxEnabled: z.boolean(),
  sandboxUrl: z.string(),
```

### 2.3 Create TanStack DB Instance

**File: `src/db/index.ts`**

```typescript
import { createDatabase } from '@tanstack/db'
import { projectSchema, taskSchema, projectSettingsSchema } from './schema'

// Create collections
export const projectsCollection = createCollection({
  name: 'projects',
  schema: projectSchema,
  primaryKey: 'id',
})

export const tasksCollection = createCollection({
  name: 'tasks',
  schema: taskSchema,
  primaryKey: 'id',
  indexes: ['projectId'], // Index for efficient querying
})

export const settingsCollection = createCollection({
  name: 'projectSettings',
  schema: projectSettingsSchema,
  primaryKey: 'projectId',
})

// Create database instance
export const db = createDatabase({
  collections: {
    projects: projectsCollection,
    tasks: tasksCollection,
    settings: settingsCollection,
  },
  // Persistence adapter (localStorage or IndexedDB)
  persistence: {
    type: 'localStorage',
    key: 'customware-portal-db',
  },
})
```

### 2.4 Create Database Hooks

**File: `src/db/hooks.ts`**
```typescript
import { useLiveQuery, useMutation } from '@tanstack/react-db'
import { db, projectsCollection, tasksCollection, settingsCollection } from './index'
import type { Project, Task, ProjectSettings } from './schema'

// Projects hooks
export function useProjects() {
  return useLiveQuery(() => 
    db.query(projectsCollection).orderBy('updatedAt', 'desc')
  )
}

export function useProject(projectId: string) {
  return useLiveQuery(() =>
    db.query(projectsCollection).where('id', '==', projectId).first()
  )
}

export function useCreateProject() {
  return useMutation({
    mutationFn: async (name: string) => {
      const now = new Date().toISOString()
      const project: Project = {
        id: Math.random().toString(36).substring(2, 9),
        name,
        createdAt: now,
        updatedAt: now,
      }
      await db.insert(projectsCollection, project)
      
      // Create default settings
      await db.insert(settingsCollection, {
        projectId: project.id,
        agentEnabled: true,
        agentModel: 'gpt-4',
        agentTemperature: 0.7,
        agentMaxTokens: 4096,
        githubEnabled: false,
        githubRepo: '',
        githubBranch: 'main',
        sandboxEnabled: false,
        sandboxUrl: '',
      })
      
      return project
    },
  })
}

export function useUpdateProject() {
  return useMutation({
    mutationFn: async ({ id, updates }: { id: string; updates: Partial<Project> }) => {
      await db.update(projectsCollection, id, {
        ...updates,
        updatedAt: new Date().toISOString(),
      })
    },
  })
}

export function useDeleteProject() {
  return useMutation({
    mutationFn: async (projectId: string) => {
      // Delete associated tasks
      const tasks = await db.query(tasksCollection).where('projectId', '==', projectId)
      for (const task of tasks) {
        await db.delete(tasksCollection, task.id)
      }
      // Delete settings
      await db.delete(settingsCollection, projectId)
      // Delete project
      await db.delete(projectsCollection, projectId)
    },
  })
}

// Tasks hooks
export function useProjectTasks(projectId: string) {
  return useLiveQuery(() =>
    db.query(tasksCollection)
      .where('projectId', '==', projectId)
      .orderBy('createdAt', 'desc')
  )
}

export function useCreateTask() {
  return useMutation({
    mutationFn: async (task: Omit<Task, 'id' | 'createdAt'>) => {
      const newTask: Task = {
        ...task,
        id: Math.random().toString(36).substring(2, 9),
        createdAt: new Date().toISOString(),
      }
      await db.insert(tasksCollection, newTask)
      return newTask
    },
  })
}

export function useUpdateTask() {
  return useMutation({
    mutationFn: async ({ id, updates }: { id: string; updates: Partial<Task> }) => {
      await db.update(tasksCollection, id, updates)
    },
  })
}

export function useDeleteTask() {
  return useMutation({
    mutationFn: async (taskId: string) => {
      await db.delete(tasksCollection, taskId)
    },
  })
}

// Settings hooks
export function useProjectSettings(projectId: string) {
  return useLiveQuery(() =>
    db.query(settingsCollection).where('projectId', '==', projectId).first()
  )
}

export function useUpdateSettings() {
  return useMutation({
    mutationFn: async ({ projectId, updates }: { projectId: string; updates: Partial<ProjectSettings> }) => {
      await db.update(settingsCollection, projectId, updates)
    },
  })
}
```

### 3.2 Task List Component Migration

**File: `src/components/TaskList.tsx`** (with TanStack DB)
```typescript
import { useProjectTasks, useCreateTask, useUpdateTask, useDeleteTask } from '../db/hooks'
import type { Task } from '../db/schema'

interface TaskListProps {
  projectId: string
}

export function TaskList({ projectId }: TaskListProps) {
  const { data: tasks = [] } = useProjectTasks(projectId)
  const createTask = useCreateTask()
  const updateTask = useUpdateTask()
  const deleteTask = useDeleteTask()

  const handleCreateTask = async () => {
    await createTask.mutateAsync({
      projectId,
      title: 'New Task',
      description: '',
      status: 'todo',
      priority: 'medium',
      completed: false,
    })
  }

  const handleToggleComplete = async (task: Task) => {
    await updateTask.mutateAsync({
      id: task.id,
      updates: { completed: !task.completed },
    })
  }

  const handleDeleteTask = async (taskId: string) => {
    await deleteTask.mutateAsync(taskId)
  }

  return (
    <div className="space-y-4">
      <button onClick={handleCreateTask} className="btn-primary">
        Add Task
      </button>
      
      {tasks.map((task) => (
        <div key={task.id} className="card p-4 flex items-center gap-4">
          <input
            type="checkbox"
            checked={task.completed}
            onChange={() => handleToggleComplete(task)}
          />
          <span className={task.completed ? 'line-through' : ''}>
            {task.title}
          </span>
          <button onClick={() => handleDeleteTask(task.id)}>
            Delete
          </button>
        </div>
      ))}
    </div>
  )
}
```

## Phase 4: Data Migration Strategy

### 4.1 Migrate Existing localStorage Data

**File: `src/db/migrate.ts`**
```typescript
import { db, projectsCollection, tasksCollection, settingsCollection } from './index'

interface LegacyState {
  projects: Array<{
    id: string
    name: string
    tasks: Array<any>
    settings: any
    createdAt: string
    updatedAt: string
  }>
}

export async function migrateFromZustand() {
  const stored = localStorage.getItem('customware-portal-storage')
  if (!stored) return

  try {
    const parsed = JSON.parse(stored)
    const state: LegacyState = parsed.state

    for (const project of state.projects) {
      // Migrate project
      await db.insert(projectsCollection, {
        id: project.id,
        name: project.name,
        createdAt: project.createdAt,
        updatedAt: project.updatedAt,
      })

      // Migrate tasks
      for (const task of project.tasks) {
        await db.insert(tasksCollection, {
          ...task,
          projectId: project.id,
        })
      }

      // Migrate settings
      if (project.settings) {
        await db.insert(settingsCollection, {
          projectId: project.id,
          ...project.settings,
        })
      }
    }

    // Remove old storage after successful migration
    localStorage.removeItem('customware-portal-storage')
    console.log('Migration completed successfully')
  } catch (error) {
    console.error('Migration failed:', error)
  }
}
```
